e.parameter.action : ''; if (action === 'count') { var formation = e.parameter.formation || ''; var date = e.parameter.date || ''; var count = countResponses(formation, date); return ContentService.createTextOutput(JSON.stringify({ status: 'ok', count: count })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ status: 'ok', count: 0, message: 'Questionnaire GetUp API' })).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var sheet = SpreadsheetApp.openById(SHEET_ID).getActiveSheet(); var template = data.template || 'general'; var templateName = data.templateName || template; var questionLabels = data.questionLabels || []; var answers = data.answers || []; // Construire les en-têtes dynamiquement à partir des labels envoyés var headers = ['Horodatage', 'Formation', 'Date formation', 'Formateur', 'Template']; for (var i = 0; i < questionLabels.length; i++) { headers.push('Q' + (i + 1) + ' - ' + questionLabels[i]); } // Créer les en-têtes si la feuille est vide if (sheet.getLastRow() === 0) { sheet.appendRow(headers); } // Construire la ligne de données var row = [ new Date().toLocaleString('fr-FR'), data.formation || '', data.date || '', data.formateur || '', templateName ]; for (var i = 0; i < answers.length; i++) { row.push(answers[i] !== null ? answers[i] : ''); } sheet.appendRow(row); // Notification email if (NOTIFY_EMAIL) { var count = countResponses(data.formation); var total = data.participants || '?'; MailApp.sendEmail({ to: NOTIFY_EMAIL, subject: '📋 Questionnaire GetUp — ' + count + '/' + total + ' réponses', htmlBody: '

Nouvelle réponse reçue !

' + '

Formation : ' + (data.formation || '') + '

' + '

Modèle : ' + templateName + '

' + '

Progression : ' + count + ' / ' + total + ' participants

' + '

Voir les réponses

' }); } return ContentService.createTextOutput(JSON.stringify({ status: 'ok', count: countResponses(data.formation) })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } } function countResponses(formation, date) { var sheet = SpreadsheetApp.openById(SHEET_ID).getActiveSheet(); var lastRow = sheet.getLastRow(); if (lastRow <= 1) return 0; // Colonnes B (formation) et C (date) → range de 2 colonnes // getDisplayValues() retourne le texte affiché (pas d'objet Date) var data = sheet.getRange(2, 2, lastRow - 1, 2).getDisplayValues(); var count = 0; for (var i = 0; i < data.length; i++) { var cellFormation = data[i][0].trim(); var cellDate = data[i][1].trim(); var matchFormation = cellFormation === formation; var matchDate = !date || cellDate === date; if (matchFormation && matchDate) count++; } return count; }
